{{- /* Markdown images. Same output as the theme's hook, plus width and height for files under static/, so the page keeps its length while lazy images load and a jump to a section lands on that section. Remote images are left as they are. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $size := dict -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if and (eq .ResourceType "image") (ne .MediaType.SubType "svg") -}} {{- $r := . -}} {{- with try ($r.Width) -}} {{- with .Value }}{{ $size = dict "width" (string $r.Width) "height" (string $r.Height) }}{{ end -}} {{- end -}} {{- end -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- else -}} {{- $file := path.Join "static" $u.Path -}} {{- if and (strings.HasPrefix $u.Path "/") (findRE `(?i)\.(png|jpe?g|gif|webp)$` $u.Path) (fileExists $file) -}} {{- with try (imageConfig $file) -}} {{- with .Value }}{{ $size = dict "width" (string .Width) "height" (string .Height) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $attributes := merge .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy") $size -}} {{- /**/ -}}
